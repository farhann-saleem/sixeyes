That means the view hasn’t been created yet—the previous query only reads it. The 100-row limit isn’t the cause.

Run this **entire block** in Supabase SQL Editor:

```sql
BEGIN;

CREATE TABLE IF NOT EXISTS public.plan_credit_limits (
  tier text PRIMARY KEY,
  avatars integer NOT NULL,
  images integer NOT NULL,
  videos integer NOT NULL,
  documentaries integer NOT NULL,
  audio integer NOT NULL,
  requests_per_minute integer NOT NULL
);

INSERT INTO public.plan_credit_limits
  (tier, avatars, images, videos, documentaries, audio, requests_per_minute)
VALUES
  ('free', 1, 5, 3, 10, 300, 6),
  ('pro', 30, 100, 30, 200, 5000, 30),
  ('premium', 300, 1000, 300, 2000, 50000, 60)
ON CONFLICT (tier) DO UPDATE SET
  avatars = EXCLUDED.avatars,
  images = EXCLUDED.images,
  videos = EXCLUDED.videos,
  documentaries = EXCLUDED.documentaries,
  audio = EXCLUDED.audio,
  requests_per_minute = EXCLUDED.requests_per_minute;

ALTER TABLE public.plan_credit_limits ENABLE ROW LEVEL SECURITY;

REVOKE ALL ON public.plan_credit_limits
  FROM PUBLIC, anon, authenticated;

GRANT SELECT ON public.plan_credit_limits TO service_role;

CREATE OR REPLACE VIEW public.user_credit_balances
WITH (security_invoker = true) AS
WITH people AS (
  SELECT
    p.*,
    to_char(now() AT TIME ZONE 'UTC', 'YYYY-MM') AS current_month,
    CASE
      WHEN p.tier IN ('pro', 'premium')
        AND p.tier_expires_at > now()
      THEN p.tier
      ELSE 'free'
    END AS effective_tier
  FROM public.profiles p
)
SELECT
  p.email,
  p.effective_tier,
  p.current_month AS month_key,
  c.credit_type,
  c.monthly_limit,
  c.used,
  CASE
    WHEN c.used IS NULL THEN NULL
    ELSE greatest(c.monthly_limit - c.used, 0)
  END AS remaining,
  l.requests_per_minute
FROM people p
JOIN public.plan_credit_limits l
  ON l.tier = p.effective_tier
CROSS JOIN LATERAL (
  VALUES
    ('avatars', l.avatars,
      CASE WHEN p.month_key = p.current_month
        THEN p.usage_avatars ELSE 0 END),
    ('images', l.images,
      CASE WHEN p.month_key = p.current_month
        THEN p.usage_images ELSE 0 END),
    ('videos', l.videos,
      CASE WHEN p.month_key = p.current_month
        THEN p.usage_videos ELSE 0 END),
    ('documentaries', l.documentaries,
      CASE WHEN p.month_key = p.current_month
        THEN p.usage_documentaries ELSE 0 END),
    ('audio', l.audio, NULL::integer)
) AS c(credit_type, monthly_limit, used);

REVOKE ALL ON public.user_credit_balances
  FROM PUBLIC, anon, authenticated;

GRANT SELECT ON public.user_credit_balances TO service_role;

COMMIT;

SELECT *
FROM public.user_credit_balances
ORDER BY email, credit_type;
```

Audio’s **used/remaining** stay `NULL` until its server ledger is migrated. Existing usage and payments remain intact.
